# bob Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `bob`, a Debian/Ubuntu system-information CLI that reports USB, wifi, bluetooth, volumes, CPU, memory, OS, and network details as a pretty table or JSON, with an interactive menu.

**Architecture:** Each category is a **collector** module with a pure `parse(raw) → data` function (unit-tested against captured command-output fixtures) and a thin `collect()` that runs the underlying tool and calls `parse`. A collector also exposes `table(data, { verbose }) → { head, rows }` for rendering. `commander` auto-registers one command per collector plus `all`; a hand-rolled `renderTable()` draws tables (no cli-table3); `@inquirer/prompts` (loaded via dynamic `import()` from CommonJS) drives the interactive menu.

**Tech Stack:** Node ≥18, CommonJS, `commander ^14`, `@inquirer/prompts ^7`, built-in `node --test` + `node:assert` for tests (no test dependencies).

## Global Constraints

- **Runtime:** Node ≥18. Target OS: Debian/Ubuntu (dev/verify machine is Ubuntu 24.04, Node v22).
- **Module system:** CommonJS (`require`/`module.exports`); **no** `"type": "module"` in package.json.
- **Dependencies:** exactly `commander ^14.0.3` and `@inquirer/prompts ^7.2.0`. No other runtime deps. **No cli-table3.** Tests use only built-in `node:test`.
- **`@inquirer/prompts` is ESM-only** — it must be loaded from CommonJS via `await import('@inquirer/prompts')`, never `require(...)`.
- **Shell safety:** all external commands run via `child_process.execFile` (never `exec`, never a shell string).
- **Graceful degradation:** a missing tool or unreadable source makes that one category report `{ available: false, reason }` — it must never throw or abort the rest of the run.
- **Collector contract:** every collector exports `{ name, title, parse, table, collect }`.
  - `parse(raw)` is pure; `raw` is a **string** for single-source collectors and an **object of named strings** for multi-source ones (wifi, bluetooth, os).
  - `collect()` resolves to `{ available: true, data }` or `{ available: false, reason }`.
  - `table(data, { verbose })` is pure and returns `{ head: string[], rows: string[][] }`.
- **Command name:** `bob`. Package/dir: `~/projects/bob`.
- **Verbose semantics:** tables are lean by default; `--verbose` adds columns/rows. JSON output is **always** the full `data` object regardless of `--verbose`.

```js
const { execFile } = require('node:child_process');

function run(cmd, args = [], { timeout = 5000 } = {}) {
  return new Promise((resolve) => {
    execFile(cmd, args, { timeout, maxBuffer: 10 * 1024 * 1024 }, (err, stdout, stderr) => {
      if (err) {
        resolve({
          ok: false,
          stdout: stdout || '',
          stderr: stderr || err.message,
          code: typeof err.code === 'number' ? err.code : null,
        });
      } else {
        resolve({ ok: true, stdout, stderr, code: 0 });
      }
    });
  });
}

async function hasCommand(cmd) {
  const res = await run('which', [cmd]);
  return res.ok && res.stdout.trim().length > 0;
}

module.exports = { run, hasCommand };
```

`src/output.js`:

```js
function formatBytes(n) {
  if (n === null || n === undefined || Number.isNaN(Number(n))) return '-';
  let bytes = Number(n);
  const units = ['B', 'KB', 'MB', 'GB', 'TB', 'PB'];
  let i = 0;
  while (bytes >= 1024 && i < units.length - 1) {
    bytes /= 1024;
    i++;
  }
  const val = i === 0 ? String(bytes) : bytes.toFixed(1);
  return `${val} ${units[i]}`;
}

function renderTable({ head, rows }) {
  const widths = head.map((h, i) =>
    Math.max(String(h).length, ...rows.map((r) => String(r[i] ?? '').length))
  );
  const line = (cells) =>
    '│ ' + cells.map((c, i) => String(c ?? '').padEnd(widths[i])).join(' │ ') + ' │';
  const sep = (l, m, r) => l + widths.map((w) => '─'.repeat(w + 2)).join(m) + r;
  const out = [sep('┌', '┬', '┐'), line(head), sep('├', '┼', '┤')];
  for (const r of rows) out.push(line(r));
  out.push(sep('└', '┴', '┘'));
  return out.join('\n');
}

function printJSON(data) {
  console.log(JSON.stringify(data, null, 2));
}

function jsonError(message, extra = {}) {
  return { error: message, ...extra };
}

module.exports = { formatBytes, renderTable, printJSON, jsonError };
```

Note: `Math.max(len, ...[])` with an empty rows array is `Math.max(len)` = `len`, so empty tables size to their headers.

```js
const { run } = require('../run');
const { formatBytes } = require('../output');

function parse(raw) {
  const json = JSON.parse(raw);
  const out = [];
  const walk = (devs, level) => {
    for (const d of devs) {
      out.push({
        name: d.name,
        size: d.size === null || d.size === undefined ? null : Number(d.size),
        type: d.type ?? null,
        fstype: d.fstype ?? null,
        mountpoint: d.mountpoint ?? null,
        model: (d.model ?? '').trim() || null,
        level,
      });
      if (Array.isArray(d.children)) walk(d.children, level + 1);
    }
  };
  walk(json.blockdevices || [], 0);
  return out;
}

function table(data, { verbose = false } = {}) {
  const head = verbose
    ? ['NAME', 'SIZE', 'TYPE', 'FSTYPE', 'MOUNTPOINT', 'MODEL']
    : ['NAME', 'SIZE', 'TYPE', 'MOUNTPOINT'];
  const rows = data.map((d) => {
    const name = '  '.repeat(d.level) + d.name;
    return verbose
      ? [name, formatBytes(d.size), d.type ?? '-', d.fstype ?? '-', d.mountpoint ?? '-', d.model ?? '-']
      : [name, formatBytes(d.size), d.type ?? '-', d.mountpoint ?? '-'];
  });
  return { head, rows };
}

async function collect() {
  const res = await run('lsblk', ['-J', '-b', '-o', 'NAME,SIZE,TYPE,FSTYPE,MOUNTPOINT,MODEL']);
  if (!res.ok) return { available: false, reason: 'lsblk unavailable (install util-linux)' };
  try {
    return { available: true, data: parse(res.stdout) };
  } catch (e) {
    return { available: false, reason: `failed to parse lsblk output: ${e.message}` };
  }
}

module.exports = { name: 'volumes', title: 'Volumes', parse, table, collect };
```

```js
const { run } = require('../run');

const num = (v) => (v === undefined || v === null || v === '' ? null : Number(v));

function parse(raw) {
  const json = JSON.parse(raw);
  const map = {};
  const walk = (entries) => {
    for (const e of entries) {
      if (e.field) map[e.field.replace(/:$/, '')] = e.data;
      if (Array.isArray(e.children)) walk(e.children);
    }
  };
  walk(json.lscpu || []);
  return {
    model: map['Model name'] ?? null,
    architecture: map['Architecture'] ?? null,
    cpus: num(map['CPU(s)']),
    threadsPerCore: num(map['Thread(s) per core']),
    coresPerSocket: num(map['Core(s) per socket']),
    sockets: num(map['Socket(s)']),
    vendor: map['Vendor ID'] ?? null,
    maxMHz: num(map['CPU max MHz']),
    minMHz: num(map['CPU min MHz']),
  };
}

function table(data, { verbose = false } = {}) {
  const fields = verbose
    ? [
        ['Model', data.model], ['Architecture', data.architecture], ['CPU(s)', data.cpus],
        ['Threads/core', data.threadsPerCore], ['Cores/socket', data.coresPerSocket],
        ['Sockets', data.sockets], ['Vendor', data.vendor],
        ['Max MHz', data.maxMHz], ['Min MHz', data.minMHz],
      ]
    : [
        ['Model', data.model], ['Architecture', data.architecture],
        ['CPU(s)', data.cpus], ['Sockets', data.sockets],
      ];
  return { head: ['FIELD', 'VALUE'], rows: fields.map(([k, v]) => [k, v ?? '-']) };
}

async function collect() {
  const res = await run('lscpu', ['-J']);
  if (!res.ok) return { available: false, reason: 'lscpu unavailable (install util-linux)' };
  try {
    return { available: true, data: parse(res.stdout) };
  } catch (e) {
    return { available: false, reason: `failed to parse lscpu output: ${e.message}` };
  }
}

module.exports = { name: 'cpu', title: 'CPU', parse, table, collect };
```

`index.js` (replaces the stub):

```js
#!/usr/bin/env node
const fs = require('node:fs');
const { program } = require('commander');
const collectors = require('./src/collectors');
const { renderTable, jsonError } = require('./src/output');

function emit(payload, opts) {
  const text = opts.json ? JSON.stringify(payload, null, 2) : payload;
  if (opts.output) fs.writeFileSync(opts.output, text + '\n');
  else console.log(text);
}

async function emitOne(collector, opts) {
  const res = await collector.collect();
  if (opts.json) {
    emit(res.available ? res.data : jsonError(res.reason), opts);
    process.exitCode = res.available ? 0 : 1;
    return;
  }
  if (!res.available) {
    console.error(`${collector.title}: ${res.reason}`);
    process.exitCode = 1;
    return;
  }
  emit(`${collector.title}\n${renderTable(collector.table(res.data, { verbose: opts.verbose }))}`, opts);
}

async function emitAll(opts) {
  if (opts.json) {
    const obj = {};
    for (const c of collectors) {
      const res = await c.collect();
      obj[c.name] = res.available ? res.data : jsonError(res.reason);
    }
    emit(obj, opts);
    return;
  }
  const blocks = [];
  for (const c of collectors) {
    const res = await c.collect();
    blocks.push(
      res.available
        ? `${c.title}\n${renderTable(c.table(res.data, { verbose: opts.verbose }))}`
        : `${c.title}\n(unavailable: ${res.reason})`
    );
  }
  emit(blocks.join('\n\n'), opts);
}

const addOpts = (cmd) =>
  cmd
    .option('--json', 'output JSON instead of a table')
    .option('-v, --verbose', 'show extra detail in tables')
    .option('-o, --output <file>', 'write output to a file');

program.name('bob').description('Debian/Ubuntu system information').version('0.1.0');
program.option('-i, --interactive', 'launch the interactive menu');

for (const c of collectors) {
  addOpts(program.command(c.name).description(`Show ${c.title.toLowerCase()}`))
    .action((opts) => emitOne(c, opts));
```

```js
const fs = require('node:fs/promises');
const nodeOs = require('node:os');
const { run } = require('../run');

function parseOsRelease(text) {
  const map = {};
  for (const line of text.split('\n')) {
    const m = line.match(/^([A-Z0-9_]+)=(.*)$/);
    if (!m) continue;
    let val = m[2].trim();
    if ((val.startsWith('"') && val.endsWith('"')) || (val.startsWith("'") && val.endsWith("'"))) {
      val = val.slice(1, -1);
    }
    map[m[1]] = val;
  }
  return map;
}

function parse(raw) {
  const rel = parseOsRelease(raw.osRelease || '');
  return {
    name: rel.PRETTY_NAME ?? rel.NAME ?? null,
    id: rel.ID ?? null,
    versionId: rel.VERSION_ID ?? null,
    kernel: (raw.kernel || '').trim() || null,
    arch: (raw.arch || '').trim() || null,
    hostname: raw.hostname || null,
  };
}

function table(data) {
  const rows = [
    ['OS', data.name], ['ID', data.id], ['Version', data.versionId],
    ['Kernel', data.kernel], ['Arch', data.arch], ['Hostname', data.hostname],
  ];
  return { head: ['FIELD', 'VALUE'], rows: rows.map(([k, v]) => [k, v ?? '-']) };
}

async function collect() {
  let osRelease = '';
  try {
    osRelease = await fs.readFile('/etc/os-release', 'utf8');
  } catch {
    return { available: false, reason: '/etc/os-release not readable' };
  }
  const kernel = (await run('uname', ['-r'])).stdout;
  const arch = (await run('uname', ['-m'])).stdout;
  return { available: true, data: parse({ osRelease, kernel, arch, hostname: nodeOs.hostname() }) };
}
```

```js
const { run } = require('../run');

function parseRfkillBlocked(rfkillRaw, type) {
  try {
    const rf = JSON.parse(rfkillRaw || '{}');
    const list = rf.rfkilldevices || rf[''] || [];
    const dev = list.find((d) => d.type === type);
    if (!dev) return null;
    return dev.soft !== 'unblocked' || dev.hard !== 'unblocked';
  } catch {
    return null;
  }
}

function parse(raw) {
  const devices = [];
  for (const line of (raw.status || '').trim().split('\n')) {
    if (!line) continue;
    const [device, type, state] = line.split(':');
    if (type === 'wifi') devices.push({ device, state: state ?? null });
  }
  let activeSsid = null;
  let signal = null;
  for (const line of (raw.wifi || '').trim().split('\n')) {
    if (!line) continue;
    const [inUse, ssid, sig] = line.split(':');
    if (inUse === '*') {
      activeSsid = ssid || null;
      signal = sig ? Number(sig) : null;
    }
  }
  return { devices, activeSsid, signal, blocked: parseRfkillBlocked(raw.rfkill, 'wlan') };
}

function table(data, { verbose = false } = {}) {
  const head = verbose
    ? ['DEVICE', 'STATE', 'SSID', 'SIGNAL', 'BLOCKED']
    : ['DEVICE', 'STATE', 'SSID'];
  const rows = data.devices.map((d) =>
    verbose
      ? [d.device, d.state ?? '-', data.activeSsid ?? '-',
         data.signal != null ? `${data.signal}%` : '-', data.blocked ? 'yes' : 'no']
      : [d.device, d.state ?? '-', data.activeSsid ?? '-']
  );
  return { head, rows };
}

async function collect() {
  const status = await run('nmcli', ['-t', '-f', 'DEVICE,TYPE,STATE', 'device', 'status']);
  if (!status.ok) return { available: false, reason: 'nmcli unavailable (install network-manager)' };
  const wifi = await run('nmcli', ['-t', '-f', 'IN-USE,SSID,SIGNAL', 'device', 'wifi']);
  const rfkill = await run('rfkill', ['-J']);
  return {
    available: true,
    data: parse({
      status: status.stdout,
      wifi: wifi.ok ? wifi.stdout : '',
      rfkill: rfkill.ok ? rfkill.stdout : '',
    }),
  };
}
```

```js
const { run } = require('../run');

function parseRfkillBlocked(rfkillRaw, type) {
  try {
    const rf = JSON.parse(rfkillRaw || '{}');
    const list = rf.rfkilldevices || rf[''] || [];
    const dev = list.find((d) => d.type === type);
    if (!dev) return null;
    return dev.soft !== 'unblocked' || dev.hard !== 'unblocked';
  } catch {
    return null;
  }
}

function parse(raw) {
  const show = raw.show || '';
  const ctrl = show.match(/Controller ([0-9A-F:]{17})/i);
  const field = (key) => {
    const m = show.match(new RegExp(`^\\s*${key}:\\s*(.*)$`, 'm'));
    return m ? m[1].trim() : null;
  };
  const controller = ctrl
    ? {
        address: ctrl[1],
        name: field('Name'),
        powered: field('Powered') === 'yes',
        discoverable: field('Discoverable') === 'yes',
      }
    : null;
  const paired = [];
  for (const line of (raw.devices || '').trim().split('\n')) {
    const m = line.match(/^Device ([0-9A-F:]{17}) (.*)$/i);
    if (m) paired.push({ mac: m[1], name: m[2].trim() });
  }
  return { controller, blocked: parseRfkillBlocked(raw.rfkill, 'bluetooth'), paired };
}

function table(data, { verbose = false } = {}) {
  const rows = [];
  if (data.controller) {
    rows.push(['Controller', data.controller.address]);
    rows.push(['Name', data.controller.name ?? '-']);
    rows.push(['Powered', data.controller.powered ? 'yes' : 'no']);
    if (verbose) rows.push(['Discoverable', data.controller.discoverable ? 'yes' : 'no']);
  }
  rows.push(['Paired devices', String(data.paired.length)]);
  if (verbose) {
    for (const p of data.paired) rows.push([`  ${p.mac}`, p.name]);
  }
  return { head: ['FIELD', 'VALUE'], rows };
}

async function collect() {
  const show = await run('bluetoothctl', ['show']);
  if (!show.ok) {
    return { available: false, reason: 'bluetoothctl unavailable (install bluez / start bluetooth)' };
  }
  const devices = await run('bluetoothctl', ['devices']);
  const rfkill = await run('rfkill', ['-J']);
  return {
    available: true,
    data: parse({
      show: show.stdout,
      devices: devices.ok ? devices.stdout : '',
      rfkill: rfkill.ok ? rfkill.stdout : '',
    }),
  };
}

module.exports = { name: 'bluetooth', title: 'Bluetooth', parse, table, collect };
```

```js
const collectors = require('./collectors');
const { renderTable } = require('./output');

async function showCollector(c) {
  const res = await c.collect();
  console.log(`\n${c.title}`);
  if (!res.available) {
    console.log(`(unavailable: ${res.reason})\n`);
    return;
  }
  console.log(renderTable(c.table(res.data, { verbose: false })) + '\n');
}

async function runInteractive() {
  const { select } = await import('@inquirer/prompts');
  const choices = [
    ...collectors.map((c) => ({ name: c.title, value: c.name })),
    { name: 'All', value: '__all__' },
    { name: 'Quit', value: '__quit__' },
  ];
  for (;;) {
    let answer;
    try {
      answer = await select({ message: 'Inspect:', choices });
    } catch {
      return; // Ctrl-C / closed prompt
    }
    if (answer === '__quit__') return;
    if (answer === '__all__') {
      for (const c of collectors) await showCollector(c);
    } else {
      await showCollector(collectors.find((c) => c.name === answer));
    }
  }
}

module.exports = { runInteractive };
```

# bob

A Debian/Ubuntu system-information CLI. Reports USB, wifi, bluetooth, volumes,
CPU, memory, OS, and network details as a pretty table or JSON, with an
interactive menu.

## Install

```bash
git clone <repo> ~/projects/bob
cd ~/projects/bob
npm install
npm link   # exposes the `bob` command
```

Requires Node ≥18 and standard Debian/Ubuntu tools: `lsblk`, `lscpu`, `free`,
`lsusb` (usbutils), `ip` (iproute2), `nmcli` (network-manager), `bluetoothctl`
(bluez), `rfkill`. Missing tools degrade gracefully — that category reports
"unavailable" and the rest still run.

## Usage

```bash
bob <category> [--json] [--verbose] [-o file]
bob all        [--json] [--verbose] [-o file]
bob -i         # interactive menu (also: bob interactive)
```

Categories: `os` `cpu` `memory` `volumes` `usb` `network` `wifi` `bluetooth`.

- `--json` — machine-readable output (always the full detail).
- `--verbose` — extra columns in table output (ignored for JSON, which is always full).
- `-o file` — write output to a file instead of stdout.

```bash
bob volumes                     # lean table of disks/partitions
```
